if(WIN32)
set(ZODIAC_CLI_PLATFORM_DIR windows)
else()
set(ZODIAC_CLI_PLATFORM_DIR posix)
endif()
# zodiac_cli_linker: link_release, separate from the executable so tests can link real release executables with it.
add_library(zodiac_cli_linker STATIC linker.cpp linker.hpp)
target_link_libraries(zodiac_cli_linker PUBLIC zodiac_core)
zodiac_target_warnings(zodiac_cli_linker)
# zodiac_cli: the whole command line front end (cli::run and the build result window), compiled once and linked into both executables
# (zodiac, console subsystem, and on Windows zodiacw, its GUI-subsystem twin) and into the tests of its parts.
add_library(zodiac_cli STATIC main.cpp templates.hpp build_command.cpp build_command.hpp build_report.cpp build_report.hpp ${PROJECT_SOURCE_DIR}/src/platform/result_window.hpp ${PROJECT_SOURCE_DIR}/src/platform/${ZODIAC_CLI_PLATFORM_DIR}/result_window.cpp)
target_compile_definitions(zodiac_cli PRIVATE "ZODIAC_VERSION=\"${PROJECT_VERSION}\"" "ZODIAC_MODULE_SUFFIX=\"${CMAKE_SHARED_LIBRARY_SUFFIX}\"")
target_link_libraries(zodiac_cli PUBLIC zodiac_cli_linker zodiac_core)
if(WIN32)
target_link_libraries(zodiac_cli PUBLIC user32 gdi32 shell32 ole32)
endif()
zodiac_target_warnings(zodiac_cli)
add_executable(zodiac ${PROJECT_SOURCE_DIR}/src/platform/process.hpp ${PROJECT_SOURCE_DIR}/src/platform/${ZODIAC_CLI_PLATFORM_DIR}/process.cpp)
set_target_properties(zodiac PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_link_libraries(zodiac PRIVATE zodiac_cli)
if(WIN32)
target_link_options(zodiac PRIVATE -municode)
# The shipped application manifest (the Segment Heap, src/platform/windows/application.manifest).
target_sources(zodiac PRIVATE $<TARGET_OBJECTS:zodiac_application_manifest>)
endif()
zodiac_target_warnings(zodiac)
if(WIN32)
# zodiacw: the same front end with a wWinMain entry (ZODIAC_GUI_ENTRY) in the GUI subsystem, so the right-click "Build with Zodiac" entry
# shows the result window without a console window behind it. It runs only from a kit layout, which building zodiac stages.
add_executable(zodiacw ${PROJECT_SOURCE_DIR}/src/platform/process.hpp ${PROJECT_SOURCE_DIR}/src/platform/windows/process.cpp)
set_target_properties(zodiacw PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_compile_definitions(zodiacw PRIVATE ZODIAC_GUI_ENTRY)
target_link_libraries(zodiacw PRIVATE zodiac_cli)
target_link_options(zodiacw PRIVATE -mwindows -municode)
target_sources(zodiacw PRIVATE $<TARGET_OBJECTS:zodiac_application_manifest>)
zodiac_target_warnings(zodiacw)
add_dependencies(zodiacw zodiac)
# The kit installs both tools into bin/ (the toolchain files that go beside them come from zodiac_kit_stage).
install(TARGETS zodiac zodiacw RUNTIME DESTINATION bin)
endif()
