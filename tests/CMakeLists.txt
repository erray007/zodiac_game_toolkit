function(zodiac_add_test name)
cmake_parse_arguments(ZT "" "" "SOURCES;LIBRARIES" ${ARGN})
add_executable(${name} ${ZT_SOURCES})
target_link_libraries(${name} PRIVATE doctest::doctest_with_main ${ZT_LIBRARIES})
zodiac_target_warnings(${name})
add_test(NAME ${name} COMMAND ${name})
endfunction()

function(zodiac_fixture_module name)
cmake_parse_arguments(ZF "LOG_ERROR;BAD_DECL" "DEP;REPORTED;ABI;STRUCT_SIZE;RESULT;EXTRA_BYTES;LICENSE;LINK_MODES" "" ${ARGN})
set(definitions "ZODIAC_FIXTURE_NAME=\"${name}\"")
if(ZF_DEP)
list(APPEND definitions "ZODIAC_FIXTURE_DEP=\"${ZF_DEP}\"")
endif()
if(ZF_REPORTED)
list(APPEND definitions "ZODIAC_FIXTURE_REPORTED=\"${ZF_REPORTED}\"")
endif()
if(DEFINED ZF_ABI)
list(APPEND definitions "ZODIAC_FIXTURE_ABI=${ZF_ABI}u")
endif()
if(DEFINED ZF_STRUCT_SIZE)
list(APPEND definitions "ZODIAC_FIXTURE_STRUCT_SIZE=${ZF_STRUCT_SIZE}u")
endif()
if(DEFINED ZF_EXTRA_BYTES)
list(APPEND definitions "ZODIAC_FIXTURE_EXTRA_BYTES=${ZF_EXTRA_BYTES}")
endif()
if(ZF_RESULT)
list(APPEND definitions "ZODIAC_FIXTURE_RESULT=${ZF_RESULT}")
endif()
if(ZF_BAD_DECL)
list(APPEND definitions "ZODIAC_FIXTURE_BAD_DECL=1")
endif()
if(ZF_LICENSE)
list(APPEND definitions "ZODIAC_FIXTURE_LICENSE=\"${ZF_LICENSE}\"")
endif()
if(ZF_LINK_MODES)
list(APPEND definitions "ZODIAC_FIXTURE_LINK_MODES=${ZF_LINK_MODES}u")
endif()
if(ZF_LOG_ERROR)
list(APPEND definitions "ZODIAC_FIXTURE_LOG_ERROR=\"fixture refuses to register\"")
endif()
zodiac_add_module(${name} SOURCES fixtures/modules/chain.cpp NO_STATIC OUTPUT_DIRECTORY "${ZODIAC_TEST_MODULE_DIR}" DEFINITIONS ${definitions})
endfunction()
