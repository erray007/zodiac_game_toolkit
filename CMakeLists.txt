# Test hooks (platform/entropy.hpp: a fixed table hash seed from ZODIAC_TABLE_HASH_SEED) for bench diagnosis on CI only. Off by default and
# in every preset; a tree with them refuses to install or package (cmake/ZodiacPackage.cmake), and test_no_test_hooks proves their absence
# from every other tree's binaries.
option(ZODIAC_TEST_HOOKS "Compile test-only hooks into the core (never for a kit or a game)" OFF)

message(WARNING "ZODIAC_TEST_HOOKS is ON: this tree is for benches and tests only; it cannot be installed or packaged.")

endif()
# The allocator experiment of Task 5c (heap-study.yml, placement-study.yml): which heap Zodiac's types and the engine allocate from in
# bench_types. Empty (the default, every preset, the kit, every game): what ships, the C runtime's heap on the process heap, which the
# application manifest makes the Segment Heap on Windows 10 2004 and later (owner decision, Task 5c fix round 1). "nt": bench_types without
# that manifest, the NT heap, for comparison. "mimalloc": Zodiac's heap calls (core/types/heap.hpp), the engine's
# (asSetGlobalMemoryFunctions) and bench_types' global operator new and delete use mimalloc (pinned in cmake/Dependencies.cmake); the owner
# decided against shipping it. Like the test hooks, an experiment tree refuses to install or package.
set(ZODIAC_BENCH_ALLOCATOR "" CACHE STRING "Bench allocator experiment: empty (as shipped), nt or mimalloc (never for a kit or a game)")
set_property(CACHE ZODIAC_BENCH_ALLOCATOR PROPERTY STRINGS "" nt mimalloc)
if(NOT ZODIAC_BENCH_ALLOCATOR STREQUAL "")
if(NOT ZODIAC_BENCH_ALLOCATOR MATCHES "^(nt|mimalloc)$")
message(FATAL_ERROR "ZODIAC_BENCH_ALLOCATOR is \"${ZODIAC_BENCH_ALLOCATOR}\"; it takes nt, mimalloc or nothing")
endif()
if(ZODIAC_BENCH_ALLOCATOR STREQUAL "nt" AND NOT WIN32)
message(FATAL_ERROR "ZODIAC_BENCH_ALLOCATOR=nt is the Windows NT heap; this tree is not Windows")
endif()
message(WARNING "ZODIAC_BENCH_ALLOCATOR is ${ZODIAC_BENCH_ALLOCATOR}: this tree is for the allocator experiment only; it cannot be installed or packaged.")
install(CODE "message(FATAL_ERROR \"This tree is configured with ZODIAC_BENCH_ALLOCATOR=${ZODIAC_BENCH_ALLOCATOR}, which must never be installed.\")")
endif()

# The application manifest of every Windows executable Zodiac ships or builds (src/platform/windows/application.manifest.in: the Segment
# Heap on Windows 10 2004 and later, and the execution level). zodiac_manifest_object(<target> <level>) configures the template for one
# requestedExecutionLevel (asInvoker or requireAdministrator) into the build tree and compiles it into the OBJECT library <target>, holding
# one RT_MANIFEST resource 1. zodiac and zodiacw link zodiac_application_manifest (asInvoker); the kit stages it as
# lib/release/zodiac_manifest.obj, which zodiac build links into every game, and zodiac_application_manifest_admin (requireAdministrator)
# as lib/release/zodiac_manifest_admin.obj, for a game that will ask to run as administrator (Task 9); bench_types links the asInvoker one,
# so the gate measures the heap games get.
enable_language(RC)
function(zodiac_manifest_object target level)
if(NOT level MATCHES "^(asInvoker|requireAdministrator)$")
message(FATAL_ERROR "zodiac_manifest_object: \"${level}\" is not an execution level (asInvoker, requireAdministrator)")
endif()
set(dir "${CMAKE_BINARY_DIR}/manifest/${level}")
set(ZODIAC_EXECUTION_LEVEL "${level}")
set(ZODIAC_MANIFEST_FILE "${dir}/application.manifest")
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/src/platform/windows/application.manifest.in" "${ZODIAC_MANIFEST_FILE}" @ONLY)
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/src/platform/windows/application.rc.in" "${dir}/application.rc" @ONLY)
add_library(${target} OBJECT "${dir}/application.rc")
set_property(SOURCE "${dir}/application.rc" APPEND PROPERTY OBJECT_DEPENDS "${ZODIAC_MANIFEST_FILE}")
endfunction()
zodiac_manifest_object(zodiac_application_manifest asInvoker)
zodiac_manifest_object(zodiac_application_manifest_admin requireAdministrator)
endif()
